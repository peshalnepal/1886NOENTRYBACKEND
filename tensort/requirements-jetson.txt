# =============================================================================
# Jetson Orin Nano (JetPack providing TensorRT 10, Python 3.10+) — pip requirements
# =============================================================================
#
# IMPORTANT: Install these INSIDE a virtualenv created with
#            `--system-site-packages` so the venv can still see the
#            JetPack-provided TensorRT, CUDA and OpenCV.
#
# DO NOT pip-install these — they come from JetPack (`sudo apt install
# nvidia-jetpack`) and MUST be the system copy:
#   - tensorrt        (built for the system Python)
#   - opencv (cv2)    (system OpenCV, built WITH GStreamer + HW decode)
#   - numpy           (already provided by the system OpenCV/JetPack stack)
#   - PyGObject / Gst / GstVideo (apt: python3-gi, gir1.2-gstreamer-1.0,
#     gir1.2-gst-plugins-base-1.0; used for native appsink capture)
#
# The pip `opencv-python` wheel has NO GStreamer support and will break
# hardware video decode on the Jetson — never install it here.
# =============================================================================

# --- Web server ---
Flask>=3.0,<4.0

# --- Database (async SQLite) ---
SQLAlchemy>=2.0,<2.1
aiosqlite>=0.20
greenlet>=3.0

# --- CUDA Python bindings (compiled against the JetPack CUDA toolkit) ---
# Needs `nvcc` on PATH at build time — see the install doc, step 5.3.
pycuda>=2025.1

# --- Config ---
# Lets `python3 main.py` read .env the same way systemd's EnvironmentFile does.
# Without it, main.py warns if .env exists and credentials are not exported.
python-dotenv>=1.0
